/*
----------------------------------------------------
Initialize Complete Weather Station
----------------------------------------------------
*/
void WeatherController_Init(void)
{
    /* Initialize ADC Driver */
    ADC_Init();

    /* Initialize LCD */
    LCD_Init();

    /* Initialize Status LED */
    LED_Init();

    /* Initialize Buzzer */
    BUZZER_Init();

    /* Initialize UART */
    UART_Init(Uart_Config);

    /* Initialize Data Logger */
    DataLogger_Init();

    /* Initialize Monitoring Module */
    Monitoring_Init();

    /* Initialize Alarm Manager */
    AlarmManager_Init();

    /* Clear LCD Screen */
    LCD_Clear();

    /* Display Welcome Message */
    LCD_WriteString("Weather Station");

    _delay_ms(1000);

    LCD_Clear();

    /* Reset Variables */
    Global_Temperature = 0;
    Global_Light = 0;
    Global_AlarmState = 0;
    Global_LogCounter = 0;
}

/*
----------------------------------------------------
Main Weather Controller Task
Call this function inside while(1)
----------------------------------------------------
*/
void WeatherController_Run(void)
{
    /* Read Temperature and Light Sensors */
    Monitoring_Update();

    /* Get Latest Temperature */
    Global_Temperature = Monitoring_GetTemperature();

    /* Get Latest Light Percentage */
    Global_Light = Monitoring_GetLight();

    /* Check Alarm Conditions */
    Global_AlarmState =
    AlarmManager_Check(Global_Temperature, Global_Light);

    /* Display Current Values on LCD */
    Monitoring_Display();

    /* Count System Cycles */
    Global_LogCounter++;

    /* Save Data Every Period */
    if(Global_LogCounter >= WEATHER_LOG_PERIOD)
    {
        DataLogger_Save
        (
            Global_Temperature,
            Global_Light,
            Global_AlarmState
        );

        /* Restart Counter */
        Global_LogCounter = 0;
    }

    /* Check UART Commands */
    UART_Handler_Task();
}
