#ifndef BITMATH_H
#define BITMATH_H

#define SetBit(Reg,BitNo)      Reg|=(1<<BitNo)
#define ClearBit(Reg,BitNo)    Reg&=~(1<<BitNo)
#define ToggleBit(Reg,BitNo)   Reg^=(1<<BitNo)
#define ReadBit(Reg,BitNo)     (Reg>>BitNo)&0x01

#define ReadFlag(Reg,FlagNo)   ReadBit(Reg,FlagNo)
#define ClearFlag(Reg,FlagNo)  SetBit(Reg,FlagNo)


#endif


#ifndef DEFINES_H
#define DEFINES_H 

#define Input   0 
#define Output  1 

#define Low     0 
#define High    1 

#define Disable 0 
#define Enable  1

#define Off     0 
#define On      1 

#define Null      (void*)0x00 
#define NullChar    '\0'
#define FlagUp       1 
#define FlagDown     0 


#define ClearRegister 0x00

#define FCPU     8000000UL

#endif
